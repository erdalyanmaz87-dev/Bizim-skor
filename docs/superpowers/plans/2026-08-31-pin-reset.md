# Kayıtlı Cihazla PIN Sıfırlama Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Oyuncunun yalnız kayıtlı cihazından, yönetici onayı olmadan dört haneli PIN'ini güvenli biçimde yenilemesini sağlamak.

**Architecture:** Yeni SECURITY DEFINER RPC, oyuncu adı ve yüksek entropili cihaz kimliğini sunucuda eşleştirir; PIN'i bcrypt ile yeniler ve eski oturumları aynı transaction içinde siler. Küçük bir istemci yardımcı modülü form doğrulama ve mesajları üretir; `index.html` yalnız formu açar ve RPC'yi çağırır.

**Tech Stack:** PostgreSQL 17, Supabase RPC/RLS, vanilla JavaScript, Node.js `node:test`, Vercel.

**Spec:** `docs/superpowers/specs/2026-08-31-pin-reset-design.md`

## Global Constraints

- Otomatik sıfırlama yalnız `players.device_id` ile tam eşleşen kayıtlı cihazda çalışır.
- PIN tam dört rakamdır ve yalnız bcrypt hash olarak saklanır.
- Başarıda bütün `friend_league_sessions` kayıtları silinir.
- Bilinmeyen oyuncu ile yanlış cihaz aynı genel destek sonucunu alır.
- Aynı cihaz 15 dakika içinde en fazla beş başarısız deneme yapabilir.
- Yeni cihaz otomatik olarak hesaba bağlanmaz.
- Mevcut giriş, profil güncelleme ve tek cihaz davranışı değişmez.

- [ ] **Step 3: Migration’ı uygula**

Migration şunları eksiksiz içermeli:

```sql
create table public.pin_reset_attempts (
  id bigint generated by default as identity primary key,
  device_hash text not null,
  attempted_at timestamptz not null default now(),
  succeeded boolean not null default false
);
alter table public.pin_reset_attempts enable row level security;
revoke all on public.pin_reset_attempts from anon, authenticated;

create or replace function public.reset_player_pin_from_registered_device(
  p_name text,
  p_device_id text,
  p_new_pin text
) returns boolean
language plpgsql
security definer
set search_path=public,extensions
as $$
declare
  v_player public.players%rowtype;
  v_device_hash text;
  v_failed_count integer;
begin
  if p_device_id is null or length(p_device_id)<20 then return false; end if;
  if p_new_pin !~ '^[0-9]{4}$' then raise exception 'PIN yalnızca 4 rakamdan oluşmalıdır'; end if;
  v_device_hash:=encode(digest(p_device_id,'sha256'),'hex');
  select count(*) into v_failed_count from public.pin_reset_attempts
   where device_hash=v_device_hash and not succeeded and attempted_at>now()-interval '15 minutes';
  if v_failed_count>=5 then raise exception 'Çok fazla deneme. 15 dakika sonra tekrar deneyin.'; end if;
  select p.* into v_player from public.players p
   where p.is_active=true
     and lower(trim(regexp_replace(p.name,'\\s+',' ','g')))=lower(trim(regexp_replace(p_name,'\\s+',' ','g')))
     and p.device_id=p_device_id
   limit 1 for update;
  if v_player.id is null then
    insert into public.pin_reset_attempts(device_hash,succeeded) values(v_device_hash,false);
    return false;
  end if;
  update public.players set pin_hash=crypt(p_new_pin,gen_salt('bf')),force_pin_once=false where id=v_player.id;
  delete from public.friend_league_sessions where player_name=v_player.name;
  insert into public.pin_reset_attempts(device_hash,succeeded) values(v_device_hash,true);
  return true;
end;
$$;
revoke all on function public.reset_player_pin_from_registered_device(text,text,text) from public;
grant execute on function public.reset_player_pin_from_registered_device(text,text,text) to anon, authenticated;
```

- [ ] **Step 3: En küçük istemci modülünü yaz**

```js
(function(root,factory){
  const api=factory();
  if(typeof module==='object'&&module.exports)module.exports=api;
  else root.BizimSkorPinReset=api;
})(typeof globalThis!=='undefined'?globalThis:this,function(){
  const normalizeName=value=>String(value||'').trim().replace(/\s+/g,' ');
  function validateResetInput({name,newPin,confirmPin}){
    const cleanName=normalizeName(name);
    if(cleanName.length<2)return{ok:false,message:'Oyuncu adını yazın.'};
    if(!/^\d{4}$/.test(String(newPin||'')))return{ok:false,message:'Yeni PIN yalnızca 4 rakamdan oluşmalıdır.'};
    if(newPin!==confirmPin)return{ok:false,message:'Yeni PIN’ler birbiriyle aynı değil.'};
    return{ok:true,name:cleanName,pin:newPin};
  }
  function resetResultMessage(success){
    return success?'PIN’iniz güncellendi. Yeni PIN’inizle giriş yapabilirsiniz.':'Bu hesap farklı bir cihazda kayıtlı. PIN sıfırlamak için yöneticiyle iletişime geçin.';
  }
  function renderResetFormMarkup(){
    return '<div id="pinResetForm" class="hide"><input id="resetName" autocomplete="username" placeholder="Oyuncu adı"><input id="resetPin" type="password" inputmode="numeric" maxlength="4" placeholder="Yeni 4 haneli PIN"><input id="resetPin2" type="password" inputmode="numeric" maxlength="4" placeholder="Yeni PIN tekrar"><button id="submitPinReset" class="p full">PIN’imi Yenile</button><div id="pinResetStatus" class="small"></div></div>';
  }
  return{validateResetInput,resetResultMessage,renderResetFormMarkup};
});
```

```js
document.getElementById('pinResetMount').innerHTML=BizimSkorPinReset.renderResetFormMarkup();
document.getElementById('openPinReset').onclick=()=>document.getElementById('pinResetForm').classList.toggle('hide');
document.getElementById('submitPinReset').onclick=async()=>{
  const button=document.getElementById('submitPinReset');
  const status=document.getElementById('pinResetStatus');
  const input=BizimSkorPinReset.validateResetInput({
    name:document.getElementById('resetName').value,
    newPin:document.getElementById('resetPin').value,
    confirmPin:document.getElementById('resetPin2').value
  });
  if(!input.ok){status.textContent=input.message;return}
  button.disabled=true;
  try{
    const q=await sb.rpc('reset_player_pin_from_registered_device',{p_name:input.name,p_device_id:ensureDeviceId(),p_new_pin:input.pin});
    if(q.error)throw q.error;
    status.textContent=BizimSkorPinReset.resetResultMessage(q.data===true);
    if(q.data===true){localStorage.removeItem('bizimSkorName');localStorage.removeItem('bizimSkorFriendToken')}
  }catch(e){status.textContent=e.message||'PIN güncellenemedi.'}
  finally{button.disabled=false}
};
```
